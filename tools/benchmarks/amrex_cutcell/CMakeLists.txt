cmake_minimum_required(VERSION 3.25)
project(cutcell_reference LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(AMReX_SPACEDIM 2 CACHE STRING "" FORCE)
set(AMReX_EB ON CACHE BOOL "" FORCE)
foreach(feature MPI OMP FORTRAN PARTICLES LINEAR_SOLVERS AMRLEVEL)
  set(AMReX_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
if(NOT AMREX_SOURCE OR NOT CARTMESH_ROOT)
  message(FATAL_ERROR "Pass AMREX_SOURCE and CARTMESH_ROOT; external source remains read-only")
endif()
add_subdirectory(${AMREX_SOURCE} amrex)
add_executable(cutcell_reference main.cpp)
target_include_directories(cutcell_reference PRIVATE ${CARTMESH_ROOT}/include)
target_link_libraries(cutcell_reference PRIVATE AMReX::amrex
  ${CARTMESH_ROOT}/build/libcartmesh2d_fv.a ${CARTMESH_ROOT}/build/libcartmesh2d.a)
